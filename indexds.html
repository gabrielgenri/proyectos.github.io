const SPREADSHEET_ID = SpreadsheetApp.getActiveSpreadsheet().getId();

function doGet(e) {
  const action = e.parameter.action || 'getRecords';
  let response = { success: false, message: 'Acción no válida' };
  
  try {
    if (action === 'getRecords') {
      response = { success: true, data: getAllRecords() };
    } else if (action === 'getMetrics') {
      response = { success: true, data: getDashboardMetrics() };
    }
  } catch (err) {
    response = { success: false, error: err.toString() };
  }
  
  return ContentService.createTextOutput(JSON.stringify(response))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  let response = { success: false, message: 'Solicitud vacía' };
  try {
    const contents = JSON.parse(e.postData.contents);
    const action = contents.action;
    
    if (action === 'createRecord') {
      response = createRecord(contents.data);
    } else if (action === 'updateRecord') {
      response = updateRecord(contents.id, contents.data);
    } else if (action === 'deleteRecord') {
      response = deleteRecord(contents.id);
    } else if (action === 'login') {
      response = authenticateUser(contents.email, contents.password);
    }
  } catch (err) {
    response = { success: false, error: err.toString() };
  }
  
  return ContentService.createTextOutput(JSON.stringify(response))
    .setMimeType(ContentService.MimeType.JSON);
}

function setupInitialSpreadsheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // 1. Hoja Usuarios
  let sheetUsuarios = ss.getSheetByName('Usuarios');
  if (!sheetUsuarios) {
    sheetUsuarios = ss.insertSheet('Usuarios');
    sheetUsuarios.appendRow(['ID_Usuario', 'Nombre', 'Email', 'PasswordHash', 'Rol', 'Estado', 'FechaCreacion']);
    sheetUsuarios.appendRow(['USR-001', 'Admin General', 'admin@sistema.com', 'admin123', 'Admin', 'Activo', new Date().toISOString()]);
  }
  
  // 2. Hoja Registros Principal
  let sheetRegistros = ss.getSheetByName('Registros');
  if (!sheetRegistros) {
    sheetRegistros = ss.insertSheet('Registros');
    sheetRegistros.appendRow(['ID', 'Fecha', 'Barrio_Zona', 'Referente_ID', 'Nombre_Solicitante', 'DNI', 'Telefono', 'Estado', 'CreadoPor', 'UltimaModificacion']);
  }
  
  // 3. Hoja Logs_Auditoria
  let sheetLogs = ss.getSheetByName('Logs_Auditoria');
  if (!sheetLogs) {
    sheetLogs = ss.insertSheet('Logs_Auditoria');
    sheetLogs.appendRow(['ID', 'FechaHora', 'Usuario_ID', 'Accion', 'Detalles', 'IP_Cliente']);
  }
}

function getAllRecords() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Registros');
  if (!sheet) return [];
  const data = sheet.getDataRange().getValues();
  const headers = data[0];
  const rows = data.slice(1);
  return rows.map(row => {
    let obj = {};
    headers.forEach((h, i) => obj[h] = row[i]);
    return obj;
  });
}

function createRecord(recordData) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Registros');
  const newId = 'REC-' + Math.floor(100000 + Math.random() * 900000);
  const now = new Date().toISOString().split('T')[0];
  
  sheet.appendRow([
    newId,
    now,
    recordData.barrio || '',
    recordData.referente || 'REF-GEN',
    recordData.nombre || '',
    recordData.dni || '',
    recordData.telefono || '',
    recordData.estado || 'Pendiente',
    recordData.usuario || 'Sistema',
    now
  ]);
  
  logAudit('CREATE', 'Nuevo registro: ' + newId);
  return { success: true, id: newId };
}

function updateRecord(id, recordData) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Registros');
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === id) {
      sheet.getRange(i + 1, 3).setValue(recordData.barrio);
      sheet.getRange(i + 1, 5).setValue(recordData.nombre);
      sheet.getRange(i + 1, 6).setValue(recordData.dni);
      sheet.getRange(i + 1, 7).setValue(recordData.telefono);
      sheet.getRange(i + 1, 8).setValue(recordData.estado);
      sheet.getRange(i + 1, 10).setValue(new Date().toISOString().split('T')[0]);
      logAudit('UPDATE', 'Actualizó registro: ' + id);
      return { success: true };
    }
  }
  return { success: false, message: 'Registro no encontrado' };
}

function deleteRecord(id) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Registros');
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === id) {
      sheet.deleteRow(i + 1);
      logAudit('DELETE', 'Eliminó registro: ' + id);
      return { success: true };
    }
  }
  return { success: false, message: 'Registro no encontrado' };
}

function authenticateUser(email, password) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Usuarios');
  if (!sheet) return { success: false, message: 'Sin usuarios registrados' };
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][2] === email && data[i][3] === password) {
      return {
        success: true,
        token: 'TOKEN-' + Math.random().toString(36).substring(2),
        user: { id: data[i][0], nombre: data[i][1], email: data[i][2], rol: data[i][4] }
      };
    }
  }
  return { success: false, message: 'Credenciales inválidas' };
}

function logAudit(accion, detalles) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Logs_Auditoria');
  if (sheet) {
    sheet.appendRow(['LOG-' + Date.now(), new Date().toISOString(), 'SISTEMA', accion, detalles, 'WEB']);
  }
}
